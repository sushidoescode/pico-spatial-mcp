## §1. HOW TO ADD A TOOL (the pattern — every task below uses it)

A tool lives in a file `src/<name>.ts` and is registered in `src/index.ts`. The simplest tools just
**return a string of generated Kotlin** (look at `src/collider.ts` and `src/environment.ts` — copy their
shape). Steps to add ANY tool:

1. **Create** `src/<name>.ts` with a `register<Name>Tools(server)` function (template in §1.1).
2. **Register** it in `src/index.ts`: add an `import` line near the other imports, and a
   `register<Name>Tools(server);` line near the other registrations (the exact spots are shown in §2.3).
3. **Build + smoke:** `npm run build && npm run smoke` → `Failures: 0`.
4. **Test the tool actually runs** with the standalone harness (§1.2).

### 1.1 The minimal tool template (copy this shape)
```ts
import type { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { z } from "zod";
import { ok, fail } from "./util.js";

export function registerExampleTools(server: McpServer) {
  server.registerTool(
    "tool_name_here",
    {
      description: "One clear sentence about what this returns.",
      inputSchema: {
        packageName: z.string().describe("Kotlin package, e.g. com.example.app.content"),
      },
    },
    async ({ packageName }) => {
      const kotlin = `package ${packageName}\n\n// ... generated code ...\n`;
      return ok(kotlin);
    },
  );
}
```

### 1.2 The standalone harness (run a tool without an MCP client)


## §2. TASK 1 (START HERE) — add `generate_scene_understanding_kotlin`

**Goal:** a tool that emits Kotlin using PICO's native scene-understanding (`sense/plane`) — the CORRECT
way to land pins on real surfaces (the native equivalent of WebXR hit-test). It returns 0 planes on the
emulator (C1) but compiles + runs now and works on the PICO Swan. This is the easiest task: it is
structurally identical to `src/environment.ts`.

### 2.1 The verified facts to embed (do NOT change these — they were confirmed on-device 2026-06-14)
- `PlaneTrackingManager.start()` — starts tracking (returns nothing).
- `PlaneTrackingManager.state` — a `TrackingState` (logs `RUNNING`).
- `suspend PlaneTrackingManager.loadAllAnchors(): Array<PlaneAnchor>` — all detected planes (0 on emulator).
- `PlaneAnchor` fields: `.transform` (position+rotation), `.boundingBoxSize: Vector2`, `.vertices: List<Vector3>`,
  `.semantics: SemanticLabelType` (FLOOR/WALL/CEILING/…), `.planeOrientation: PlaneOrientation`.
- Imports: `com.pico.spatial.sense.plane.PlaneTrackingManager`, `com.pico.spatial.sense.plane.PlaneAnchor`.
- The `sense` dependency is ALREADY in the app (`implementation(libs.sense)`), so this Kotlin compiles.

### 2.2 Create `src/sense.ts` with EXACTLY this content
```ts
import type { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { z } from "zod";
import { ok } from "./util.js";

/**
 * generate_scene_understanding_kotlin — the CORRECT native surface-pin path (sense/plane).
 * VERIFIED on emulator-5554 2026-06-14: PlaneTrackingManager.start()/state/loadAllAnchors() compile +
 * run; the emulator returns 0 planes (no sensors). Lands real-surface pins on a PICO Swan headset.
 */
export function registerSenseTools(server: McpServer) {
  server.registerTool(
    "generate_scene_understanding_kotlin",
    {
      description:
        "Emit drop-in Kotlin (SceneUnderstanding.kt) that uses PICO native scene-understanding (sense/plane) to land markers on REAL detected surfaces (floor/wall/ceiling) — the native equivalent of WebXR hit-test. NOTE: the EMULATOR returns 0 planes (no sensors); this code compiles + runs now and snaps pins on a PICO Swan headset. Pair with a fixed-distance fallback for emulator demos.",
      inputSchema: {
        packageName: z.string().describe("Kotlin package, e.g. com.example.app.content"),
      },
    },
    async ({ packageName }) => {
      const kotlin = [
        `package ${packageName}`,
        "",
        "import android.util.Log",
        "import com.pico.spatial.core.math.Vector3",
        "import com.pico.spatial.sense.plane.PlaneAnchor",
        "import com.pico.spatial.sense.plane.PlaneTrackingManager",
        "import kotlinx.coroutines.Dispatchers",
        "import kotlinx.coroutines.withContext",
        "",
        "/**",
        " * Generated by pico-spatial-mcp generate_scene_understanding_kotlin.",
        " * !!! VERIFY ON: PICO Swan headset. The EMULATOR returns 0 planes (sense has no emulator data",
        " * source — confirmed 2026-06-14: state=RUNNING but loadAllAnchors()=0). nearestPlaneHit returns",
        " * null on the emulator -> keep a fixed-distance fallback for emulator demos.",
        " */",
        "object SceneUnderstanding {",
        '    private const val TAG = "SceneUnderstanding"',
        "",
        "    /** Start plane tracking. Call once after the scene loads. */",
        "    fun start() {",
        "        runCatching { PlaneTrackingManager.start(); Log.i(TAG, \"plane tracking ${PlaneTrackingManager.state}\") }",
        '            .onFailure { Log.e(TAG, "plane start failed: ${it.message}") }',
        "    }",
        "",
        "    /** All currently detected planes (empty on the emulator). */",
        "    suspend fun planes(): List<PlaneAnchor> =",
        "        runCatching { withContext(Dispatchers.IO) { PlaneTrackingManager.loadAllAnchors().toList() } }",
        '            .getOrElse { Log.e(TAG, "loadAllAnchors failed: ${it.message}"); emptyList() }',
        "",
        "    /** Log every detected plane (use to confirm scene-understanding works on hardware). */",
        "    suspend fun logPlanes() {",
        "        val ps = planes()",
        '        Log.i(TAG, "planes detected: ${ps.size}")',
        "        ps.forEachIndexed { i, p ->",
        '            Log.i(TAG, "  plane[$i] semantics=${p.semantics} orient=${p.planeOrientation} bbox=${p.boundingBoxSize} verts=${p.vertices.size}")',
        "        }",
        "    }",
        "",
        "    /** Raycast the gaze ray vs the detected planes; nearest hit point + plane normal, or null.",
        "     *  NOTE: javap-confirm PlaneAnchor.transform accessors before trusting on hardware. */",
        "    suspend fun nearestPlaneHit(origin: Vector3, dir: Vector3, maxDist: Float = 8f): Pair<Vector3, Vector3>? {",
        "        val ps = planes()",
        '        if (ps.isEmpty()) { Log.w(TAG, "nearestPlaneHit: 0 planes (emulator? use fixed-distance fallback)"); return null }',
        "        var best = maxDist; var pt: Vector3? = null; var nrm = Vector3.UP",
        "        for (p in ps) {",
        "            val c = p.transform.position",
        "            val n = p.transform.rotation.rotateVector(Vector3.FORWARD).normalize()",
        "            val denom = Vector3.dot(dir, n)",
        "            if (kotlin.math.abs(denom) < 1e-5f) continue",
        "            val t = Vector3.dot(c.minus(origin), n) / denom",
        "            if (t in 0.01f..best) { best = t; pt = origin.plus(dir.times(t)); nrm = n }",
        "        }",
        "        return pt?.let { it to nrm }",
        "    }",
        "}",
        "",
      ].join("\n");
      return ok(kotlin);
    },
  );
}
```
